Why flutter_file_downloader?
Unlike pure Dart networking utilities (like HTTP or Dio), this package routes the request directly through the native Android background architecture. It automatically hooks into the native Android notification layout engine to show a system-level progress bar and allows clicking on the finished notification to view the file.

Android Configuration

To ensure your app has permission to write to public spaces and push progress updates to the notification drawer, modify your android/app/src/main/AndroidManifest.xml file.

Inside the <manifest> tag, add the following parameters:

<uses-permission android:name="android.permission.INTERNET"/>
<!-- Required for Android 13+ to show system notifications -->
<uses-permission android:name="android.permission.POST_NOTIFICATIONS"/>

<!-- Required only if you support Android 9 or lower -->
<uses-permission android:name="android.permission.WRITE_EXTERNAL_STORAGE" android:maxSdkVersion="28"/>


Implementation Code

The process involves two steps:

    Generate and cache the local PDF inside your app's temporary folder.

    Pass the file path or a mock server endpoint to the download manager to transfer it directly to the native Downloads/ partition with notifications enabled.

import 'dart:io';
import 'package:flutter/material.dart';
import 'package:path_provider/path_provider.dart';
import 'package:pdf/widgets.dart' as pw;
import 'package:permission_handler/permission_handler.dart';
import 'package:flutter_file_downloader/flutter_file_downloader.dart';

class DocumentDownloadService {
  
  /// Step 1: Generates a raw PDF layout and caches it temporarily
  Future<File> generateSamplePDF() async {
    final pdf = pw.Document();

    pdf.addPage(
      pw.Page(
        build: (pw.Context context) {
          return pw.Center(
            child: pw.Text(
              "Generated Statement",
              style: pw.TextStyle(fontSize: 40, fontWeight: pw.FontWeight.bold),
            ),
          );
        },
      ),
    );

    // Get temp directory to store file before copying to system downloads
    final outputDir = await getTemporaryDirectory();
    final file = File("${outputDir.path}/generated_document.pdf");
    await file.writeAsBytes(await pdf.save());
    return file;
  }

  /// Step 2: Requests system authorization and hands off file to the background downloader
  Future<void> downloadPdfWithNotification({
    required Function(double progress) onProgressUpdated,
    required Function(File? file) onDownloadComplete,
  }) async {
    // 1. Request Notification permission for Android 13+
    if (await Permission.notification.request().isGranted) {
      
      // 2. Build the local document cache
      File localFile = await generateSamplePDF();
      
      // 3. Trigger native file download pipeline with progress tracking
      // Note: In case your file is generated via a backend API directly, 
      // replace the local URI string with your secure API endpoint.
      await FileDownloader.downloadFile(
        url: Uri.file(localFile.path).toString(),
        name: "My_Generated_File.pdf",
        notificationType: NotificationType.all, // Shows progress, success, & failure states
        downloadDestination: DownloadDestinations.publicDownloads, // Saves to device Downloads folder
        onProgress: (String? fileName, double progress) {
          // Progress returns values from 0.0 to 100.0
          onProgressUpdated(progress);
        },
        onDownloadCompleted: (String path) {
          onDownloadComplete(File(path));
        },
        onDownloadError: (String error) {
          debugPrint("Download pipeline encountered an error: $error");
        }
      );
    } else {
      debugPrint("System Notification permission was denied by the user.");
    }
  }
}



ui example of implementation
class DownloadScreen extends StatefulWidget {
  const DownloadScreen({Key? key}) : super(key: key);

  @override
  State<DownloadScreen> createState() => _DownloadScreenState();
}

class _DownloadScreenState extends State<DownloadScreen> {
  final DocumentDownloadService _downloadService = DocumentDownloadService();
  double _currentProgress = 0.0;
  bool _isProcessing = false;

  void _triggerDownload() async {
    setState(() {
      _isProcessing = true;
      _currentProgress = 0.0;
    });

    await _downloadService.downloadPdfWithNotification(
      onProgressUpdated: (progress) {
        setState(() {
          _currentProgress = progress;
        });
      },
      onDownloadComplete: (file) {
        setState(() {
          _isProcessing = false;
        });
        if (file != null) {
          ScaffoldMessenger.of(context).showSnackBar(
            SnackBar(content: Text("PDF saved successfully to: ${file.path}")),
          );
        }
      },
    );
  }

  @override
  Widget build(BuildContext context) {
    return Scaffold(
      appBar: AppBar(title: const Text("Document Center")),
      body: Center(
        child: _isProcessing
            ? Column(
                mainAxisAlignment: MainAxisAlignment.center,
                children: [
                  CircularProgressIndicator(value: _currentProgress / 100),
                  const SizedBox(height: 16),
                  Text("Downloading: ${_currentProgress.toStringAsFixed(0)}%"),
                ],
              )
            : ElevatedButton(
                onPressed: _triggerDownload,
                child: const Text("Generate & Download PDF"),
              ),
      ),
    );
  }
}
